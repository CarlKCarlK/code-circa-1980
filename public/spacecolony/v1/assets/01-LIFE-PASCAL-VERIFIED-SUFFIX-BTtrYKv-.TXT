	FUNC NAYB(X,Y);
		VAR A,B,NAY:INTEGER;
		BEGIN
			NAY:=0;
			FOR A:=X-1 TO X+1 DO FOR B:=Y-1 TO Y+1 DO
					NAY:=NAY-(POINT(A MOD 64,B MOD 48) AND
					((A<>X) OR (B<>Y)));
					NAYB:=NAY
			END;
		FUNC PNT(X,Y);
			CONST ALT=31743;
			VAR BIT,LOC,M:INTEGER;
			BEGIN
				BIT:=Y MOD 3;
				LOC:=ALT + X + (Y DIV 3);
				M:=MEM(LOC) SHR (BIT + BIT);
				PNT:= (M AND 1)=1
				END;
		PROC PLT(X,Y,C);
			CONST ALT=31743;
			VAR BIT,LOC,MASK:INTEGER;
			BEGIN
				MASK:=3;
				BIT:=Y MOD 3;
				LOC:=ALT + X + (Y DIV 3);
				MASK:=MASK SHL (BIT+BIT);
				MEM(LOC):=MEM(LOC) AND (1-MASK) AND 63 + 128;
				IF C THEN BEGIN
					CELLS:=CELLS+1;
					MEM(LOC):=MEM(LOC) OR MASK
					END
				END;
	BEGIN
		CELLS:=0;
		FOR X:=0 TO 63 DO FOR Y:=0 TO 47 DO BEGIN 
			CASE NAYB(X,Y) OF 
				0,1:PLT(X,Y,0);
				2:PLT(X,Y,POINT(X,Y));
				3:PLT(X,Y,1);
				4,5,6,7,8:PLT(X,Y,0)
				END
			END;
		MOVE(31743,15360,1024)
		END;
BEGIN 
  INIT;
  REPEAT LIFE UNTIL CELLS=0;
